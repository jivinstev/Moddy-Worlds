[other-repos]
# A link to any other repository of this account whose name starts with "moddy-": this repository (moddy-worlds) and
# the core (ModdyTheWishWatch, no hyphen, so the rule never sees it) are the only ones public text may point at.
# moddy-the-wish-watch is the core's own former name (it redirects there, 2026-11-23): earlier commits carry it.
re:\bjivinstev/moddy-(?!worlds\b|the-wish-watch\b)[A-Za-z0-9_.-]+

[plan-ids]
# A step id from a private plan (a capital T, digits, a dot and digits; or HD, a dash and digits): a comment names what a thing does, not the step that built it.
# Case-sensitive (a capital T or HD), so a lower-case "t1.5" or an "hd-video" is fine.
re:(?-i:(?<![\w.-])T\d{1,2}[A-Z]?\.\d{1,2}[a-z]?(?!\w|\.\d))
re:(?-i:(?<![\w-])HD-\d{1,2}(?![\w]))

[player-ids]
# A literal player id that looks real (version 3 is an offline-mode player, version 4 an online account). A test makes
# its own with UUID.randomUUID(), or uses an obviously synthetic one (all zeros but the last digits, and the like).
re:\b[0-9a-f]{8}-[0-9a-f]{4}-[34][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b
allow:^(0{8}|1{8}|f{8}|12345678|deadbeef|cafebabe)-

[server-manifest]
# A world manifest or rules file carrying the half only a hosted server needs: its hostnames, ports, machine,
# camera pool, spend caps, access policy or the agent's brief. The level's world files hold gameplay only.
re:"(hostnames|ports|house|camera_vm_pool|camera_host|camera_pool|spend_cap_usd_day|spend_bucket_usd|proxy_online_mode|wish_slots_camera_on_vm|brief)"\s*:
path:*.env

[local-paths]
# A home directory or a cloud session's paths in committed text.
re:/home/user/
re:/Users/[A-Za-z0-9._-]+/
re:/tmp/claude-

[neoforge-versions]
# NeoForge numbers 26.x by four parts (the Minecraft version, then a build), which the built-in IPv4 rule reads as an
# address. Each version a branch pins is allowed here by its exact text, one line per version (as the core's list does).
allow:^26\.1\.2\.130$
allow:^26\.3\.0\.117$
